$ git remote
$ git remote add origin ~/server/bike-notes.git
$ git remote -v
origin	/home/sam/server/bike-notes.git (fetch)
origin	/home/sam/server/bike-notes.git (push)
$ git config --get-regexp remote
remote.origin.url /home/sam/server/bike-notes.git
remote.origin.fetch +refs/heads/*:refs/remotes/origin/*
$ git remote rename origin home
$ git config --get-regexp remote
remote.home.url /home/sam/server/bike-notes.git
remote.home.fetch +refs/heads/*:refs/remotes/home/*
$ git remote add origin ~/club/bike-notes.git
$ git remote -v

home	/home/sam/server/bike-notes.git (push)

origin	/home/sam/club/bike-notes.git (push)
